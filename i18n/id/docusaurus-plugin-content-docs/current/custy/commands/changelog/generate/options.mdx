## Ringkasan

<CommandOption
  options={[
    {
      name: "--commit-msg-file",
      shorthand: "-cmsg",
      description:
        "Override template commit message untuk pending-commit parsing dan classification. File harus tersedia.",
      defaultValue: ".config/custy/templates/commit-message.txt",
      example: "--commit-msg-file templates/release-message.txt",
    },
    {
      name: "--force / --no-force",
      shorthand: "-f / -F",
      description:
        "Aktifkan atau nonaktifkan forced regeneration untuk invocation ini.",
      defaultValue: "false unless configured otherwise",
      example: "--force",
    },
  ]}
/>
