## Ringkasan

<CommandExample
  examples={[
    {
      title: "Gunakan Project Configuration",
      description: "Jalankan kebijakan cleanup proyek.",
      command: `custy cleanup branches`,
      result: "Mengevaluasi branch berdasarkan konfigurasi aktif.",
    },
    {
      title: "Batasi Prefix",
      description: "Pilih release branches.",
      command: `custy cleanup branches --include-prefix release/`,
      result: "Hanya branch dengan prefix release/ dipertimbangkan.",
    },
    {
      title: "Semua Merge Status",
      description: "Override merge status.",
      command: `custy cleanup branches --merge-status all`,
      result: "Merge status tidak lagi membatasi kandidat.",
    },
    {
      title: "Lebih Lama dari 30 Hari",
      description: "Gunakan usia relatif.",
      command: `custy cleanup branches --max-age 30d`,
      result: "Memilih branch yang melewati batas usia.",
    },
    {
      title: "Sebelum Tanggal Tertentu",
      description: "Gunakan tanggal absolut.",
      command: `custy cleanup branches --before 2026-07-01`,
      result: "Memilih branch sebelum tanggal tersebut.",
    },
    {
      title: "Preview Cleanup",
      description: "Tinjau kandidat tanpa menghapus.",
      command: `custy --dry-run cleanup branches`,
      result: "Menampilkan rencana cleanup.",
    },
    {
      title: "Debug Cleanup",
      description: "Tambahkan diagnostik.",
      command: `custy --debug cleanup branches`,
      result: "Menjalankan cleanup dengan output debug.",
    },
  ]}
/>

<InfoBox variant="danger" title="Tinjau Konfigurasi">
  Jangan gunakan `--merge-status all` tanpa memahami dampaknya. Kombinasikan filter
  secara hati-hati dan lakukan dry run.
</InfoBox>
