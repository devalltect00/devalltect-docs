## Ringkasan

<InfoBox variant="tip" title="Confirm the Exact Tag">
  Mulai dengan `custy --dry-run tag --tag v1.4.0`.
</InfoBox>

<CommandExample
  examples={[
    {
      title: "Dry Run",
      command: "custy --dry-run tag --tag v1.4.0",
      description: "Tinjau exact tag tanpa membuat reference.",
      result: "Menampilkan planned Tag profile.",
    },
    {
      title: "Project Defaults",
      command: "custy tag",
      description: "Gunakan resolved configuration.",
      result: "Membentuk versi dan membuat local tag.",
    },
    {
      title: "Explicit SemVer",
      command: "custy tag --strategy semver --tag v1.4.0",
      description: "Gunakan exact SemVer.",
      result: "Membuat v1.4.0.",
    },
    {
      title: "Explicit SemVer Release Candidate",
      command: "custy tag --strategy semver --tag v1.4.0-rc.1",
      description: "Gunakan approved explicit lifecycle tag.",
      result: "Membuat v1.4.0-rc.1.",
    },
    {
      title: "Inline Message",
      command: 'custy tag --tag v1.4.0 --tag-message "Release v1.4.0"',
      description: "Gunakan inline tag message.",
      result: "Membuat annotated tag dengan pesan tersebut.",
    },
  ]}
/>

<InfoBox variant="danger" title="Do Not Force an Unexplained Failure">
  Jangan gunakan `--force-tag` sebelum memahami conflict dan publication state.
</InfoBox>
