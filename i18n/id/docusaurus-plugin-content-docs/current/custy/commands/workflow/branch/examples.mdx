## Ringkasan

<InfoBox variant="danger" title="Documentation Examples Only">
  Examples dapat mencapai known pipeline blocker dan tidak boleh digunakan dalam CI/CD.
</InfoBox>
<CommandExample
  examples={[
    {
      title: "Branch Check",
      command:
        "custy --dry-run workflow branch --check_transition --from-branch develop --to-branch main",
      result: "Menyusun transition request sebelum known blocker.",
    },
    {
      title: "Tag Check",
      command: "custy --dry-run workflow branch -ct --from-tag v1.2.0 --to-tag v1.3.0",
      result: "Menyusun tag transition request.",
    },
    {
      title: "Diagnostics",
      command: "custy --debug --log-level debug workflow branch --check_transition",
      result: "Menampilkan diagnostics untuk development investigation.",
    },
  ]}
/>
