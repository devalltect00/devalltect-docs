remove_separators = ["---"]
remove_headers = false
remove_hash_lines = false

remove_metadata = false
remove_keywords = [

  "- _(Nothing yet)_",

<InfoBox variant="warning" title="Keyword Matching Bersifat Case-Sensitive">
  Gunakan exact phrases dan ubah satu cleaning rule setiap kali. Broad removals dapat
  menghasilkan valid Markdown yang diam-diam kehilangan useful release context.
</InfoBox>
